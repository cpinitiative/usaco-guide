## Explanation

For every word, we have to verify whether we can form it by taking
at most one letter from each of the given blocks. Since we are
unaware of which block to choose first, we will enumerate over
their permutations, as there are only $4! = 24$ permutations of blocks. For each
permutation, we will greedily search through the blocks in that
order: whenever we get a match, we search for the next letter;
otherwise, we skip it.

## Implementation

**Time Complexity:** $\mathcal{O}(N)$

```cpp
#include <algorithm>
#include <functional>
#include <iostream>

int main() {
	int n;
	std::cin >> n;
	std::array<std::array<int, 26>, 4> cnt{0};
	std::string block;
	for (int i = 0; i < 4; i++) {
		std::cin >> block;
		for (char c : block) { cnt[i][c - 'A']++; }
	}

	auto is_possible = [&](std::string word) {
		std::array<int, 4> perm = {0, 1, 2, 3};
		do {
			for (int i = 0, j = 0; i < 4; i++) {
				bool ok = false;
				if (cnt[perm[i]][word[j] - 'A']) { j++; }
				if (j == (int)word.size()) { return true; }
			}
		} while (std::next_permutation(begin(perm), end(perm)));
		return false;
	};

	for (int i = 1; i <= n; i++) {
		std::string word;
		std::cin >> word;
		std::cout << (is_possible(word) ? "YES" : "NO") << '\n';
	}
}
```

```java
import java.io.*;
import java.util.*;

public class Blocks {
	// cnt[i][j] = number of times letter j appears on block i
	static int[][] cnt = new int[4][26];

	// checks whether the given word can be formed using the four blocks
	static boolean isPossible(String word) {
		int[] perm = {0, 1, 2, 3};

		do {
			int j = 0;

			for (int i = 0; i < 4; i++) {
				if (j == word.length()) { return true; }

				if (cnt[perm[i]][word.charAt(j) - 'A'] > 0) { j++; }
			}

			if (j == word.length()) { return true; }

		} while (nextPermutation(perm));

		return false;
	}

	// generate the next lexicographical permutation of a
	static boolean nextPermutation(int[] a) {
		int i = a.length - 2;

		while (i >= 0 && a[i] >= a[i + 1]) { i--; }

		if (i < 0) { return false; }

		int j = a.length - 1;
		while (a[j] <= a[i]) { j--; }

		int temp = a[i];
		a[i] = a[j];
		a[j] = temp;

		for (int l = i + 1, r = a.length - 1; l < r; l++, r--) {
			temp = a[l];
			a[l] = a[r];
			a[r] = temp;
		}

		return true;
	}

	public static void main(String[] args) throws IOException {
		Scanner sc = new Scanner(System.in);

		int n = sc.nextInt();

		for (int i = 0; i < 4; i++) {
			String block = sc.next();

			for (char c : block.toCharArray()) { cnt[i][c - 'A']++; }
		}

		for (int i = 0; i < n; i++) {
			String word = sc.next();

			System.out.println(isPossible(word) ? "YES" : "NO");
		}

		sc.close();
	}
}
```
