class Solution {
  public:
	vector<vector<int>> matrixBlockSum(vector<vector<int>> &mat, int K) {
		int n = mat.size(), m = mat[0].size();
		vector<vector<int>> arr(n);
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) { arr[i].push_back(mat[i][j]); }
		}
		vector<vector<int>> dp(n + 1);
		for (int i = 0; i <= n; i++) {
			for (int j = 0; j <= m; j++) { dp[i].push_back(0); }
		}
		for (int i = 1; i <= n; i++) {
			for (int j = 1; j <= m; j++) {
				dp[i][j] =
				    dp[i - 1][j] + dp[i][j - 1] - dp[i - 1][j - 1] + arr[i - 1][j - 1];
			}
		}
		vector<vector<int>> answer;
		for (int i = 1; i <= n; i++) {
			vector<int> cur;
			for (int j = 1; j <= m; j++) {
				int a = max(1, i - K), b = max(1, j - K);
				int c = min(n, i + K), d = min(m, j + K);
				int blockSum =
				    dp[c][d] - dp[c][b - 1] - dp[a - 1][d] + dp[a - 1][b - 1];
				cur.push_back(blockSum);
			}
			answer.push_back(cur);
		}
		return answer;
	}
};

class Solution {
    public int[][] matrixBlockSum(int[][] mat, int K) {
        
        int m = mat.length, n = mat[0].length;
        int[][] ans = new int[m][n];

        // sum[i][j] is the sum of all elements from the rectangle (0, 0, i, j)
        int[][] sum = new int[m + 1][n + 1];

		// build the 2d prefix sum matrix
        for(int r = 1; r <= m; r++) {
            for(int c = 1; c <= n; c++) {
                sum[r][c] = mat[r - 1][c - 1] + sum[r - 1][c] + sum[r][c - 1] - sum[r - 1][c - 1];
            }
        }
		// find the block sum for every cell of the matrix 
        for(int r = 0; r < m; r++) {
            for(int c = 0; c < n; c++) {

				// finding the block boundaries
                int r1 = Math.max(0, r - K), c1 = Math.max(0, c - K);
                int r2 = Math.min(m - 1, r + K), c2 = Math.min(n - 1, c + K);
                
				// convert matrix indices to prefix sum indices
				r1++; c1++; r2++; c2++;

				// calculating rectangle sum using the prefix sums
                ans[r][c] = sum[r2][c2] + sum[r1 - 1][c1 - 1] - sum[r2][c1 - 1] - sum[r1 - 1][c2];
            }
        }

        return ans;

    }
}

class Solution:
    def matrixBlockSum(self, mat, K):

        m, n = len(mat), len(mat[0])
        ans = [[0] * n for _ in range(m)]

        # sum[i][j] is the sum of all elements from the rectangle (0, 0, i, j)
        prefix = [[0] * (n + 1) for _ in range(m + 1)]

        # build the 2D prefix sum matrix
        for r in range(1, m + 1):
            for c in range(1, n + 1):
                prefix[r][c] = (
                    mat[r - 1][c - 1]
                    + prefix[r - 1][c]
                    + prefix[r][c - 1]
                    - prefix[r - 1][c - 1]
                )

        # find the block sum for every cell
        for r in range(m):
            for c in range(n):

                # find block boundaries
                r1 = max(0, r - K)
                c1 = max(0, c - K)
                r2 = min(m - 1, r + K)
                c2 = min(n - 1, c + K)

                # convert matrix indices to prefix sum indices
                r1 += 1
                c1 += 1
                r2 += 1
                c2 += 1

                # calculating rectangle sum using prefix sums
                ans[r][c] = (
                    prefix[r2][c2]
                    + prefix[r1 - 1][c1 - 1]
                    - prefix[r2][c1 - 1]
                    - prefix[r1 - 1][c2]
                )
        return ans
