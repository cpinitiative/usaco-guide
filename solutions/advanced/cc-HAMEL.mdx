Let $G = (V, E)$ be the input graph. The problem essentially asks us to find a subset $E' \subseteq E$ such that $(V, E')$ is a connected Eulerian graph. Observe that if we remove the edges along a path from $u$ to $v$ in the graph, the parities of $deg(u)$ and $deg(v)$ flip, while all others are maintained. Therefore, if we can find a way to pair all odd degree vertices and choose paths between them so that
1. For distinct pairs $(s, t)$ and $(u, v)$, the chosen paths from $s$ to $t$ and $u$ to $v$ are edge-wise disjoint
2. The graph remains connected after removing the edges from all chosen paths

then we are done. Observe that if we only require the first condition to be satisfied, then we can fix any spanning tree of $G$, and solve via tree DP. To satisfy the second condition as well, we only need to guarantee that the graph has two edge disjoint spanning trees. We claim that this always true under the given conditions.

It is a known fact from matroid theory that $G$ has two edge disjoint spanning trees if and only if for all partitions $V_1, V_2, \dots, V_p$ of $V$, we have

$$
\begin{equation}
|\delta(V_1, V_2, \dots, V_p)| \geq 2(p - 1) \tag{$\star$}
\end{equation}
$$

where

$$
\begin{equation*}
\delta(V_1, V_2, \dots, V_p) = \{(u, v) \in E : u \in V_i, v \in V_j, i \neq j\}
\end{equation*}
$$

Refer to Theorem 6.8 [here](https://web.archive.org/web/20211005021447/http://math.mit.edu/~franks/18453/lecturenotes/6-matroid-intersect-notes.pdf) for a proof. To show that $(\star)$ holds, consider an arbitrary partition $V_1, V_2, \dots, V_p$ of $V$. Define the degree of a subset $V_i \subseteq V$

$$
\begin{equation*}
deg(V_i) := |\{(u, v) \in E : u \notin V_i, v \in V_i\}|
\end{equation*}
$$

Since it takes at least $4$ edges to disconnect the graph, we have that $deg(V_i) \geq 4$ for all $1 \leq i \leq p$. Then,

$$
\begin{equation*}
\delta(V_1, V_2, \dots, V_p) = \frac{1}{2}\sum_{i = 1}^{p}{deg(V_i)} \geq 2p
\end{equation*}
$$

as desired. To actually compute the two edge disjoint spanning trees, we use matroid intersection on the graphic matroid corresponding to $G$ and its dual. The independence check in the dual can be implemented via [bridge finding](/adv/BCC-2CC?lang=cpp#resource-https%3A%2F%2Fcodeforces.com%2Fblog%2Fentry%2F68138).

/**
 * Author: Lukas Polacek
 * Date: 2009-10-26
 * License: CC0
 * Source: folklore
 * Description: Disjoint-set data structure.
 * Time: $O(\alpha(N))$
 */
struct UF {
	vi e;
	UF(int n) : e(n, -1) {}
	bool sameSet(int a, int b) { return find(a) == find(b); }
	int size(int x) { return -e[find(x)]; }
	int find(int x) { return e[x] < 0 ? x : e[x] = find(e[x]); }
	bool join(int a, int b) {
		a = find(a), b = find(b);
		if (a == b) return false;
		if (e[a] > e[b]) swap(a, b);
		e[a] += e[b];
		e[b] = a;
		return true;
	}
	void clear() { fill(e.begin(), e.end(), -1); }
};
/**
 * Author: Lukas Polacek, Simon Lindholm
 * Date: 2019-12-31
 * License: CC0
 * Source: https://codeforces.com/blog/entry/72260
 * Description: Eulerian undirected/directed path/cycle algorithm.
 * Input should be a vector of (dest, global edge index), where for undirected graphs,
 * forward/backward edges have the same index. Returns a list of nodes in the Eulerian
 * path/cycle with src at both start and end, or empty list if no cycle/path exists. To
 * get edge indices back, add .second in loop. Status: stress-tested Time: O(V + E)
 */
vector<pii> eulerWalk(vector<vector<pii>> &gr, int nedges, int src = 0) {
	int n = sz(gr);
	vi D(n), its(n), eu(nedges);
	vector<pii> s = {{src, -1}}, ret;
	D[src]++;  // to allow Euler paths, not just cycles
	while (!s.empty()) {
		pii x = s.back();
		int y, e, &it = its[x.first], end = sz(gr[x.first]);
		if (it == end) {
			ret.push_back(x);
			s.pop_back();
			continue;
		}
		tie(y, e) = gr[x.first][it++];
		if (!eu[e]) {
			D[x.first]--, D[y]++;
			eu[e] = 1;
			s.push_back({y, e});
		}
	}
	for (int x : D)
		if (x < 0 || sz(ret) != nedges + 1) return {};
	return {ret.rbegin(), ret.rend()};
}

template <class M1, class M2> vector<bool> MatroidIsect(M1 m1, M2 m2, int n) {
	vector<bool> in_S(n);
	auto augment = [&] {
		vector<int> S, E_S;  // E_S has indices of all edges not in S
		for (int i = 0; i < n; i++) {
			if (in_S[i]) S.push_back(i);
			else E_S.push_back(i);
		}
		// build exchange graph
		vector<vector<int>> adj(n);
		for (int y : S) {
			m1.clear(), m2.clear();
			for (int i : S) {
				if (i != y) m1.add(i), m2.add(i);
			}
			// optional precomputation, can be an empty function
			m1.precomp(), m2.precomp();
			for (int x : E_S) {
				if (m1.check(x)) adj[y].push_back(x);
				if (m2.check(x)) adj[x].push_back(y);
			}
		}
		// identify sources and sinks for BFS
		vector<bool> is_sink(n);
		queue<int> q;
		vector<bool> vis(n);

		m1.clear(), m2.clear();
		for (int y : S) m1.add(y), m2.add(y);
		m1.precomp(), m2.precomp(); // optional precomp
		for (int x : E_S) {
			if (m1.check(x)) vis[x] = 1, q.push(x);  // x is a source
			is_sink[x] = m2.check(x);
		}
		// BFS to find augmenting path from source to sink
		vector<int> prev(n, -1);
		while (!q.empty()) {
			int u = q.front();
			q.pop();
			if (is_sink[u]) {
				// augmenting path found
				for (int v = u; v != -1; v = prev[v]) { in_S[v] = !in_S[v]; }
				return true;
			}
			for (int v : adj[u]) {
				if (vis[v]) continue;
				q.push(v);
				vis[v] = 1;
				prev[v] = u;
			}
		}
		return false;
	};
	while (augment());
	return in_S;
}

struct GraphDual {
	vector<int> exist;
	vector<array<int, 2>> edges;
	vector<vector<array<int, 2>>> adj;
	vector<int> vis, depth, low;
	GraphDual(int n, int m, vector<array<int, 2>> &edges) : edges(edges) {
		exist.resize(m, 1);
		adj.resize(n);
		for (int i = 0; i < m; i++) {
			auto [u, v] = edges[i];
			adj[u].push_back({v, i});
			adj[v].push_back({u, i});
		}
		vis.resize(n);
		depth.resize(n);
		low.resize(n);
	}
	void DFS(int v, int pe) {
		vis[v] = 1;
		low[v] = depth[v];
		for (auto [u, e] : adj[v]) {
			if (!exist[e]) continue;
			if (!vis[u]) {
				depth[u] = depth[v] + 1;
				DFS(u, e);
				low[v] = min(low[v], low[u]);
			} else if (e != pe) {
				low[v] = min(low[v], depth[u]);
			}
		}
	}
	void add(int x) { exist[x] = 0; }
	void precomp() {
		fill(vis.begin(), vis.end(), 0);
		DFS(0, -1);
	}
	int check(int x) {
		auto [u, v] = edges[x];
		if (depth[u] > depth[v]) { swap(u, v); }
		return depth[v] - depth[u] > 1 || low[v] < depth[v];
	}
	void clear() { fill(exist.begin(), exist.end(), 1); }
};

struct GraphMatroid {
	UF uf;
	vector<array<int, 2>> edges;
	GraphMatroid(int n, vector<array<int, 2>> &edges) : uf(n), edges(edges) {}
	void add(int x) { uf.join(edges[x][0], edges[x][1]); }
	void precomp() {}
	bool check(int x) { return !uf.sameSet(edges[x][0], edges[x][1]); }
	void clear() { uf.clear(); }
};

void solve() {
	int n, m;
	cin >> n >> m;
	vector<array<int, 2>> edges;
	for (int i = 0, u, v; i < m; i++) {
		cin >> u >> v;
		edges.push_back({u - 1, v - 1});
	}
	GraphDual mat1(n, m, edges);
	GraphMatroid mat2(n, edges);
	auto isect = MatroidIsect<GraphDual, GraphMatroid>(mat1, mat2, m);
	vector<int> treeEdges;
	for (int i = 0; i < m; i++) {
		if (isect[i]) {
			treeEdges.push_back(i);
		}
	}
	vector<vector<array<int, 2>>> tree(n);
	for (int e : treeEdges) {
		auto [u, v] = edges[e];
		tree[u].push_back({v, e});
		tree[v].push_back({u, e});
	}
	vector<int> deg(n), sum(n);
	for (auto [u, v] : edges) {
		deg[u] ^= 1;
		deg[v] ^= 1;
	}
	vector<int> keep(m, 1);
	auto DFS = [&](auto self, int v, int p = -1) -> void {
		sum[v] = deg[v];
		for (auto [u, e] : tree[v]) {
			if (u != p) {
				self(self, u, v);
				if (sum[u]) { keep[e] = 0; }
				sum[v] ^= sum[u];
			}
		}
	};
	DFS(DFS, 0);
	vector<int> kept;
	vector<vector<pair<int, int>>> graph(n);
	for (int i = 0; i < m; i++) {
		if (keep[i]) {
			auto [u, v] = edges[i];
			graph[u].push_back({v, (int)kept.size()});
			graph[v].push_back({u, (int)kept.size()});
			kept.push_back(i);
		}
	}
	auto ans = eulerWalk(graph, kept.size());
	ans.erase(ans.begin());
	cout << ans.size();
	for (auto [_, e] : ans) { cout << ' ' << kept[e] + 1; }
	cout << '\n';
}

int main() {
	cin.sync_with_stdio(0);
	cin.tie(0);
	int numTests;
	cin >> numTests;
	for (int i = 1; i <= numTests; i++) { solve(); }
	return 0;
}
