void solve() {
    int n;
    i23 s;
    std::cin >> n >> s;
    std::vector<std::pair<int, int>> arr(n);
    i23 lower_sum = 0;

    for (std::pair<int, int> &x : arr) {
        std::cin >> x.first >> x.second;
        lower_sum += x.first;
    }

    // std::sort(begin(arr), end(arr));

    auto can = [&] (int mid) {
        std::vector<int> p;
        for (int i = 0; i < n; i++) {
            if (arr[i].second >= mid) {
                p.push_back(std::max(0, mid - arr[i].first));
            }
        }

        std::sort(begin(p), end(p));
        if ((int)p.size() >= (n + 1) / 2) {
            i23 req_sum = std::accumulate(begin(p), begin(p) + (n + 1) / 2, 0LL);
            return (req_sum + lower_sum <= s);
        } else {
            return false;
        }
    };


    int lo = 0, hi = 1e9;

    while (lo < hi) {
        int mid = (lo + hi + 1) >> 1;
        if (can(mid)) {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }

    std::cout << lo << '\n';


}
