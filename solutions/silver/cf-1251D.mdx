#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>

int main() {
	int test_num;
	std::cin >> test_num;
	
	for (int t = 0; t < test_num; t++) {
		
		int n;
		long long s;
		std::cin >> n >> s;
		std::vector<std::pair<int, int>> arr(n);
		long long lower_sum = 0;

		for (std::pair<int, int> &x : arr) {
			std::cin >> x.first >> x.second;
			lower_sum += x.first;
		}

		auto is_possible = [&](int mid) {
			std::vector<int> p;
			for (int i = 0; i < n; i++) {
				if (arr[i].second >= mid) { p.push_back(std::max(0, mid - arr[i].first)); }
			}

			std::sort(begin(p), end(p));

			if ((int)p.size() >= (n + 1) / 2) {
				long long req_sum = std::accumulate(begin(p), begin(p) + (n + 1) / 2, 0LL);
				return (req_sum + lower_sum <= s);
			} else {
				return false;
			}
		};

		int lo = 0, hi = 1e9;

		while (lo < hi) {
			int mid = (lo + hi + 1) >> 1;
			if (is_possible(mid)) {
				lo = mid;
			} else {
				hi = mid - 1;
			}
		}

		std::cout << lo << '\n';
	}
	
}
