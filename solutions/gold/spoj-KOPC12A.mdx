## Explanation

Let's define $f(x)$ as the total cost required to make
all buildings have height $x$. For the $i$-th building to reach
height $x$, the cost contributed by it will be:

$$
g_i(x) = |x - h_i| \cdot c_i
$$

Therefore, the total reconstruction cost is:
$$
f(x) = \sum_{i=1}^{n} g_i(x) = \sum_{i=1}^{n} |x - h_i| \cdot c_i
$$

Our final goal is to find the value $x$ which minimizes $f(x)$.

We know that $g_i(x)$ is convex, as it is an absolute value function.
Since the sum of convex functions is also convex, $f(x)$ is also convex.

A convex function is also unimodal; hence, $f(x)$ is unimodal (i.e., it strictly decreases, reaches a minimum, then strictly increases). This allows us to apply ternary search over the range of possible heights to find the minimum cost.


## Implementation

**Time Complexity:** $\mathcal{O}(N \log{H})$, where $H$ is the maximum building height.

```cpp
#include <iostream>
#include <vector>

const int MAX_H = 10000;

int main() {

	int test_num;
	std::cin >> test_num;
	for (int t = 0; t < test_num; t++) {

		int n;
		std::cin >> n;
		std::vector<int> heights(n);
		std::vector<int> costs(n);

		for (int &x : heights) { std::cin >> x; }

		for (int &x : costs) { std::cin >> x; }

		/** @return total reconstruction cost at height x*/
		const auto get_cost = [&](int mid) -> long long {
			long long res = 0;
			for (int i = 0; i < n; i++) {
				res += std::abs(mid - heights[i]) * costs[i];
			}
			return res;
		};

		int lo = 0, hi = MAX_H;
		while (hi - lo > 3) {
			int mid1 = lo + (hi - lo) / 3;
			int mid2 = hi - (hi - lo) / 3;
			if (get_cost(mid1) > get_cost(mid2)) {
				lo = mid1;
			} else {
				hi = mid2;
			}
		}

		long long min_cost = get_cost(lo);
		for (int i = lo + 1; i <= hi; i++) {
			min_cost = std::min(min_cost, get_cost(i));
		}

		std::cout << min_cost << '\n';
	}
}
```

import java.io.*;
import java.util.*;

public class Main {

	static int[] heights, costs;
	static int n;

	/** @return total reconstruction cost at height x*/
	static long getCost(int mid) {
		long res = 0;
		for (int i = 0; i < n; i++) {
			res += (long)Math.abs(mid - heights[i]) * costs[i];
		}
		return res;
	}

	public static void main(String[] args) throws Exception {
		Kattio io = new Kattio();

		int testNum = io.nextInt();
		final int MAX_H = 10000;

		for (int t = 0; t < testNum; t++) {
			n = io.nextInt();
			heights = new int[n];
			costs = new int[n];

			for (int i = 0; i < n; i++) { heights[i] = io.nextInt(); }
			for (int i = 0; i < n; i++) { costs[i] = io.nextInt(); }

			int lo = 0, hi = MAX_H;
			while (hi - lo > 3) {
				int mid1 = lo + (hi - lo) / 3;
				int mid2 = hi - (hi - lo) / 3;
				if (getCost(mid1) > getCost(mid2)) {
					lo = mid1;
				} else {
					hi = mid2;
				}
			}

			long minCost = getCost(lo);
			for (int i = lo + 1; i <= hi; i++) {
				minCost = Math.min(minCost, getCost(i));
			}

			io.println(minCost);
		}

		io.close();
	}

	// BeginCodeSnip{Kattio}
}

for _ in range(int(input())):
