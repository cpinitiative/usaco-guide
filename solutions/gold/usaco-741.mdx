## Solution 2

Sort suffixes.

## Implementation

**Time Complexity:** $\mathcal{O}(NM^2\log N)$ solution:

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	freopen("cownomics.in", "r", stdin);
	freopen("cownomics.out", "w", stdout);

	int N, M;
	cin >> N >> M;

	vector<string> g(2 * N);
	for (auto &s : g) cin >> s;

	int ans = INT_MAX;

	for (int i = 0; i < M; i++) {
		vector<int> v(2 * N);

		iota(v.begin(), v.end(), 0);

		auto common = [&](int x, int y) {
			int ind = i;
			while (ind < M && g[x][ind] == g[y][ind]) ind++;
			return ind;
		};

		auto cmp = [&](int x, int y) {
			int ind = common(x, y);
			return ind < M && g[x][ind] < g[y][ind];
		};

		sort(v.begin(), v.end(), cmp);

		int mx = i;

		for (int j = 0; j < 2 * N - 1; j++) {
			if (v[j] / N != v[j + 1] / N) { mx = max(mx, common(v[j], v[j + 1])); }
		}

		if (mx == M) break;

		ans = min(ans, mx + 1 - i);
	}

	cout << ans << "\n";
}
```

**Time Complexity:** $\mathcal{O}(NM^2)$ solution:

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
	ios::sync_with_stdio(false);
	cin.tie(nullptr);

	freopen("cownomics.in", "r", stdin);
	freopen("cownomics.out", "w", stdout);

	int N, M;
	cin >> N >> M;

	vector<string> g(2 * N);
	for (auto &s : g) cin >> s;

	vector<int> suf(2 * N);

	iota(suf.begin(), suf.end(), 0);

	int ans = INT_MAX;

	for (int i = M - 1; i >= 0; i--) {
		vector<int> tmp[26];

		for (int t : suf) { tmp[g[t][i] - 'A'].push_back(t); }

		suf.clear();

		for (int j = 0; j < 26; j++) {
			suf.insert(suf.end(), tmp[j].begin(), tmp[j].end());
		}

		auto common = [&](int x, int y) {
			int ind = i;
			while (ind < M && g[x][ind] == g[y][ind]) ind++;
			return ind;
		};

		int mx = i;

		for (int j = 0; j < 2 * N - 1; j++) {
			if (suf[j] / N != suf[j + 1] / N) {
				mx = max(mx, common(suf[j], suf[j + 1]));
			}
		}

		if (mx < M) { ans = min(ans, mx + 1 - i); }
	}

	cout << ans << "\n";
}
```

## Solution 3

First, compute prefix hashes for every spotted and plain cow so that the hash of any substring can be retrieved in $\mathcal{O}(1)$ time. Since any substring length greater than a valid distinguishing length is also valid, we can binary search on the minimum valid substring length $L$. For a fixed length $L$, we iterate over all possible starting positions $i$ ($0 \le i \le M - L$), store the substring hashes of all spotted cows in a map, and check if any plain cow shares a hash in that map. If at least one starting position has no collisions between the two groups, length $L$ is valid.
**Time Complexity:** $\mathcal{O}(NM\log N\log M)$ solution:

## Implementation

```cpp
#include <algorithm>
#include <chrono>
#include <iostream>
#include <map>
#include <random>
#include <set>
#include <vector>

using namespace std;

#define int long long

mt19937_64 rng((int)chrono::steady_clock::now().time_since_epoch().count());
int M = ((int)1 << 61) - 1;
int b = uniform_int_distribution<int>(2, M - 1)(rng);

class hashedstring {
  private:
	vector<int> pow;
	vector<int> p_hash;

  public:
	void setup(const string &s) {
		p_hash.resize(s.size() + 1);
		pow.resize(s.size() + 1);
		pow[0] = 1;
		for (int i = 1; i < s.size() + 1; i++) {
			pow[i] = ((__int128)pow[i - 1] * b) % M;
		}
		p_hash[0] = 0;
		for (int i = 0; i < s.size(); i++) {
			p_hash[i + 1] = ((__int128)p_hash[i] * b + s[i]) % M;
		}
	}
	int gethash(int s, int e) {
		int ret = p_hash[e + 1] - ((__int128)p_hash[s] * pow[e - s + 1]) % M;
		return (ret + M) % M;
	}
};

signed main() {
	ios::sync_with_stdio(false);
	cin.tie(NULL);
	freopen("cownomics.in", "r", stdin);
	freopen("cownomics.out", "w", stdout);
	int n, m;
	cin >> n >> m;
	vector<hashedstring> s(n);
	vector<hashedstring> p(n);
	for (int i = 0; i < n; i++) {
		string a;
		cin >> a;
		s[i].setup(a);
	}
	for (int i = 0; i < n; i++) {
		string a;
		cin >> a;
		p[i].setup(a);
	}
	int lo = 1;
	int hi = m;
	while (lo < hi) {
		int mid = (lo + hi) / 2;
		bool finalworks = false;
		for (int i = 0; i < m - mid + 1; i++) {
			bool works = true;
			map<int, bool> m;
			for (int j = 0; j < n; j++) { m[s[j].gethash(i, i + mid - 1)] = true; }
			for (int j = 0; j < n; j++) {
				if (m[p[j].gethash(i, i + mid - 1)] == true) works = false;
			}
			if (works) finalworks = true;
		}
		if (finalworks) hi = mid;
		else lo = mid + 1;
	}
	cout << lo << endl;
}
```
